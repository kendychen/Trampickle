package core

// Xác định IP thật của khách. Đây là nền của mọi giới hạn tần suất: sai chỗ
// này thì rate limit, khoá đăng nhập và nhật ký đều ghi nhầm người.
//
// Bản cũ lấy phần tử ĐẦU của X-Forwarded-For. Phần tử đầu là do khách gửi —
// bot chỉ cần đổi header mỗi request là mọi giới hạn thành số 0, và mã đơn
// vốn theo số thứ tự thì dò ra được cả danh sách khách.
//
// Luật mới, theo thứ tự:
//   1. Request đến từ dải Cloudflare  → tin CF-Connecting-IP.
//   2. Request đến từ loopback/mạng nội bộ và có -tin-proxy → tin X-Real-IP,
//      không có thì lấy phần tử CUỐI của X-Forwarded-For (phần nginx tự nối).
//   3. Còn lại → RemoteAddr.

import (
	"net"
	"net/http"
	"strings"
)

// TinProxy: chỉ bật khi thật sự có nginx đứng trước (cờ -tin-proxy).
var TinProxy bool

// Dải IP Cloudflare, nhúng cứng theo https://www.cloudflare.com/ips/
// (bản chép ngày 2026-09-07). Cố ý không tải qua mạng lúc khởi động: một lần
// Cloudflare đổi endpoint là dịch vụ không lên được, đắt hơn cái nó giải quyết.
// Cloudflare đổi dải rất hiếm; khi đổi thì sửa tay ở đây rồi build lại.
var daiCloudflare = []string{
	"173.245.48.0/20", "103.21.244.0/22", "103.22.200.0/22", "103.31.4.0/22",
	"141.101.64.0/18", "108.162.192.0/18", "190.93.240.0/20", "188.114.96.0/20",
	"197.234.240.0/22", "198.41.128.0/17", "162.158.0.0/15", "104.16.0.0/13",
	"104.24.0.0/14", "172.64.0.0/13", "131.0.72.0/22",
	"2400:cb00::/32", "2606:4700::/32", "2803:f800::/32", "2405:b500::/32",
	"2405:8100::/32", "2a06:98c0::/29", "2c0f:f248::/32",
}

var mangCF []*net.IPNet

func init() {
	for _, s := range daiCloudflare {
		_, m, err := net.ParseCIDR(s)
		if err != nil {
			// Dải viết sai trong hằng số là lỗi lập trình, không phải lỗi
			// vận hành. Bỏ qua im lặng thì Cloudflare hoá ra không được tin
			// mà không ai biết.
			panic("dai Cloudflare sai cu phap: " + s)
		}
		mangCF = append(mangCF, m)
	}
}

func laCloudflare(ip net.IP) bool {
	for _, m := range mangCF {
		if m.Contains(ip) {
			return true
		}
	}
	return false
}

func ipCua(r *http.Request) string {
	host, _, err := net.SplitHostPort(r.RemoteAddr)
	if err != nil {
		host = r.RemoteAddr
	}
	ip := net.ParseIP(host)
	if ip == nil {
		return host
	}

	if laCloudflare(ip) {
		if v := strings.TrimSpace(r.Header.Get("CF-Connecting-IP")); v != "" {
			return v
		}
	}

	if TinProxy && (ip.IsLoopback() || ip.IsPrivate()) {
		if v := strings.TrimSpace(r.Header.Get("X-Real-IP")); v != "" {
			return v
		}
		if h := r.Header.Get("X-Forwarded-For"); h != "" {
			phan := strings.Split(h, ",")
			return strings.TrimSpace(phan[len(phan)-1])
		}
	}

	return host
}
```

# Việc bảo mật hạ tầng — Kendy chạy trên VPS

Làm theo thứ tự. Mỗi mục có lệnh kiểm chứng; không qua được thì dừng, đừng

## 6. Cloudflare (bấm trên web)

## 7. Backup data/ — quan trọng hơn mọi mục trên

`data/` chỉ tồn tại trên VPS. Deploy không đụng tới. Mất là mất hết đơn hàng,
sổ tiền, bài viết.

    sudo mkdir -p /opt/backup && sudo chmod 700 /opt/backup
    sudo tee /opt/tramvot/backup.sh > /dev/null <<'EOF'
    #!/bin/sh
    set -e
    NGAY=$(date +%F)
    tar czf - -C /opt/tramvot data \
      | gpg --batch --yes --symmetric --cipher-algo AES256 \
            --passphrase-file /root/.backup-pass \
            -o /opt/backup/data-$NGAY.tar.gz.gpg
    find /opt/backup -name 'data-*.gpg' -mtime +30 -delete
    EOF
    sudo chmod 700 /opt/tramvot/backup.sh

Đặt mật khẩu backup (gõ một chuỗi dài, lưu vào trình quản lý mật khẩu):

    sudo sh -c 'printf "%s" "MAT-KHAU-DAI-O-DAY" > /root/.backup-pass'
    sudo chmod 600 /root/.backup-pass

Chạy hàng ngày lúc 3h sáng:

    sudo crontab -e
    # thêm dòng:
    0 3 * * * /opt/tramvot/backup.sh

package core

// Chặn dò mật khẩu ở /dang-nhap.
//
// Đếm theo CẢ IP lẫn tên tài khoản, vì mỗi cách một mình đều có lỗ:
//   - chỉ theo IP  : botnet nhiều IP dò một tài khoản vẫn lọt.
//   - chỉ theo tên : kẻ xấu cố tình gõ sai để khoá tài khoản Kendy (DoS).
//
// Cách thoát DoS: khoá theo tên chỉ chặn IP LẠ. IP đã từng đăng nhập thành
// công vào tài khoản đó vẫn vào được, nên Kendy ngồi ở tiệm không bao giờ bị
// người ngoài khoá.
//
// Giữ trong RAM như phiên đăng nhập: restart là quên hết. Chấp nhận — kẻ dò
// không làm cho tiến trình restart được, và đổi lại không có file trạng thái
// nào phải dọn.

import (
	"sync"
	"time"
)

const (
	saiToiDaIP  = 5
	saiToiDaTen = 10
	khoaDau     = 15 * time.Minute
	khoaTran    = 4 * time.Hour
)

type demSai struct {
	sai     int
	lanKhoa int
	moLuc   time.Time
}

var (
	khoaMu  sync.Mutex
	khoaIP  = map[string]*demSai{}
	khoaTen = map[string]*demSai{}
	// ipQuen: tên tài khoản -> tập IP đã từng đăng nhập thành công.
	ipQuen = map[string]map[string]bool{}
)

// hanKhoa: 15 phút, rồi 30, 60, 120, trần 4 giờ.
func hanKhoa(lan int) time.Duration {
	d := khoaDau
	for i := 1; i < lan; i++ {
		d *= 2
		if d >= khoaTran {
			return khoaTran
		}
	}
	return d
}

// choPhepThu trả về false kèm thời gian còn phải chờ.
func choPhepThu(ip, ten string) (bool, time.Duration) {
	khoaMu.Lock()
	defer khoaMu.Unlock()
	now := time.Now()

	if d := khoaIP[ip]; d != nil && now.Before(d.moLuc) {
		return false, d.moLuc.Sub(now)
	}
	if d := khoaTen[ten]; d != nil && now.Before(d.moLuc) && !ipQuen[ten][ip] {
		return false, d.moLuc.Sub(now)
	}
	return true, 0
}

func ghiThuSai(ip, ten string) {
	khoaMu.Lock()
	defer khoaMu.Unlock()
	dem(khoaIP, ip, saiToiDaIP)
	dem(khoaTen, ten, saiToiDaTen)
}

// dem tăng bộ đếm và khoá khi chạm ngưỡng. Gọi khi đang giữ khoaMu.
func dem(m map[string]*demSai, k string, nguong int) {
	d := m[k]
	if d == nil {
		d = &demSai{}
		m[k] = d
	}
	// Khoá cũ đã hết hạn thì bắt đầu đếm lại từ đầu, nhưng lanKhoa giữ
	// nguyên — kẻ quay lại dò tiếp phải chờ lâu hơn lần trước.
	if !d.moLuc.IsZero() && time.Now().After(d.moLuc) {
		d.sai = 0
		d.moLuc = time.Time{}
	}
	d.sai++
	if d.sai >= nguong {
		d.lanKhoa++
		d.moLuc = time.Now().Add(hanKhoa(d.lanKhoa))
		d.sai = 0
	}
}

func ghiThuDung(ip, ten string) {
	khoaMu.Lock()
	defer khoaMu.Unlock()
	delete(khoaIP, ip)
	if khoaTen[ten] != nil {
		khoaTen[ten].sai = 0
	}
	if ipQuen[ten] == nil {
		ipQuen[ten] = map[string]bool{}
	}
	// Giới hạn 20 IP mỗi tài khoản, khỏi phình vô hạn nếu Kendy dùng 4G đổi
	// IP liên tục. Quá thì xoá sạch, lần sau nhớ lại từ đầu.
	if len(ipQuen[ten]) >= 20 {
		ipQuen[ten] = map[string]bool{}
	}
	ipQuen[ten][ip] = true
}

package core

// Chuỗi middleware bọc ngoài mux.
//
// Vì sao không nhét vào từng handler: hơn 100 route đã đăng ký, rải tay là
// chắc chắn sót, và mỗi route thêm sau này lại phải nhớ. Vì sao NewMux giữ
// nguyên chữ ký: hơn chục test đang nhận *http.ServeMux, đổi là vỡ hết.

import (
	"context"
	"net/http"
)

type khoaCtx int

const (
	ctxNonce khoaCtx = iota
	ctxCSRF
)

// NewHandler: mux + middleware. main.go dùng cái này; test trang cũ vẫn dùng
// NewMux trực tiếp.
func NewHandler(public bool) http.Handler {
	var h http.Handler = NewMux(public)
	h = boCSRF(h)
	h = boHeader(h)
	return h
}

// boHeader nằm ngoài cùng để cả trang lỗi 403/429 cũng có header bảo mật.
func boHeader(h http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		nonce := maNgauNhien(16)

		w.Header().Set("X-Content-Type-Options", "nosniff")
		w.Header().Set("X-Frame-Options", "DENY")
		w.Header().Set("Referrer-Policy", "strict-origin-when-cross-origin")
		w.Header().Set("Permissions-Policy", "camera=(), microphone=(), geolocation=(), payment=()")

		// HSTS chỉ khi thật sự có HTTPS. Gửi sớm là trình duyệt nhớ, và máy
		// nhà chạy HTTP sẽ không vào được nữa mà không hiểu vì sao.
		if HTTPSBat {
			w.Header().Set("Strict-Transport-Security", "max-age=31536000; includeSubDomains")
		}

		// style-src chấp nhận 'unsafe-inline': trong core/ui có hàng trăm
		// thuộc tính style="...". Sửa hết là việc lớn, rủi ro vỡ giao diện
		// cao, mà lợi ích thấp hơn hẳn script-src. script-src thì dùng nonce
		// thật, không nới.
		w.Header().Set("Content-Security-Policy",
			"default-src 'self'; "+
				"script-src 'self' 'nonce-"+nonce+"'; "+
				"style-src 'self' 'unsafe-inline'; "+
				"img-src 'self' data:; "+
				"font-src 'self'; "+
				"connect-src 'self'; "+
				"object-src 'none'; "+
				"base-uri 'self'; "+
				"form-action 'self'; "+
				"frame-ancestors 'none'")

		h.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), ctxNonce, nonce)))
	})
}

func nonceCua(r *http.Request) string {
	s, _ := r.Context().Value(ctxNonce).(string)
	return s
}

package core

// Chống CSRF bằng token, không chỉ dựa vào SameSite=Lax nữa.
//
// SameSite=Lax vẫn giữ làm lớp hai. Nó chặn được phần lớn, nhưng: Lax cho
// phép GET điều hướng mang cookie, vài trình duyệt cũ hiểu Lax khác nhau, và
// nếu sau này có route nào nhận GET gây tác dụng phụ thì Lax không cứu.
//
// Hai kiểu token, vì có hai loại người dùng:
//   - Đã đăng nhập : token = HMAC(khoá server, mã phiên). Không cần lưu gì
//     thêm, và token tự chết cùng phiên.
//   - Khách        : không có phiên nên không có gì để buộc token vào.
//     Dùng double-submit: cookie và ô ẩn trong form phải khớp nhau. Trang
//     khác không đọc được cookie của mình nên không đặt đúng ô ẩn được.

import (
	"crypto/hmac"
	"crypto/sha256"
	"crypto/subtle"
	"encoding/hex"
	"net/http"
)

const (
	tenTruongCSRF = "_csrf"
	tenCookieCSRF = "tv_csrf"
)

// khoaCSRF sinh lúc khởi động, giữ trong RAM. Restart là token cũ hết hiệu
// lực — cùng vòng đời với phiên đăng nhập, nên hai thứ không lệch nhau.
var khoaCSRF = func() []byte {
	b, _ := hex.DecodeString(maNgauNhien(32))
	return b
}()

func tokenTuPhien(maPhien string) string {
	m := hmac.New(sha256.New, khoaCSRF)
	m.Write([]byte(maPhien))
	return hex.EncodeToString(m.Sum(nil))
}

// tokenCSRF: token mà template phải nhúng vào form của request này.
func tokenCSRF(r *http.Request) string {
	s, _ := r.Context().Value(ctxCSRF).(string)
	return s
}

func khopHang(a, b string) bool {
	return a != "" && subtle.ConstantTimeCompare([]byte(a), []byte(b)) == 1
}

func boCSRF(h http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// Token của phiên nếu đã đăng nhập; nếu không thì token khách.
		var tok string
		if c, err := r.Cookie(tenCookie); err == nil && c.Value != "" {
			tok = tokenTuPhien(c.Value)
		} else {
			c, err := r.Cookie(tenCookieCSRF)
			if err == nil && c.Value != "" {
				tok = c.Value
			} else {
				tok = maNgauNhien(16)
			}
			// Đặt lại mỗi lần: khách vào trang lần đầu chưa có cookie, mà
			// template cần một giá trị để nhúng ngay trong lần render này.
			http.SetCookie(w, &http.Cookie{
				Name:     tenCookieCSRF,
				Value:    tok,
				Path:     "/",
				HttpOnly: false,
				Secure:   HTTPSBat,
				SameSite: http.SameSiteLaxMode,
				MaxAge:   12 * 60 * 60,
			})
		}

		switch r.Method {
		case http.MethodPost, http.MethodPut, http.MethodPatch, http.MethodDelete:
			// ParseMultipartForm cho form có file; FormValue tự gọi
			// ParseMultipartForm khi Content-Type là multipart.
			if !khopHang(r.FormValue(tenTruongCSRF), tok) {
				http.Error(w, "Phiên làm việc đã cũ. Tải lại trang rồi gửi lại giúp em.",
					http.StatusForbidden)
				return
			}
		}

		h.ServeHTTP(w, r.WithContext(ctxDatCSRF(r, tok)))
	})
}
```

type MucNhatKy struct {
	Luc      string `json:"luc"`
	Ai       string `json:"ai"`
	IP       string `json:"ip"`
	Viec     string `json:"viec"`
	Duong    string `json:"duong,omitempty"`
	DoiTuong string `json:"doi_tuong,omitempty"`
	KetQua   string `json:"ket_qua"`

// truongCam: giá trị của những trường này KHÔNG BAO GIỜ vào nhật ký. Nhật ký
// là file đọc được, để lộ mật khẩu ở đây là tự tạo ra chỗ rò mới.
var truongCam = map[string]bool{
	"mat_khau":       true,
	"mat_khau_moi":   true,
	"mat_khau_cu":    true,
	"_csrf":          true,
	"gemini_api_key": true,
	"resend_api_key": true,
	"totp":           true,
	"ma_totp":        true,
	"ma_du_phong":    true,
}

var nhatKyMu sync.Mutex

func thuMucNhatKy() string { return P("data/nhat-ky") }

func GhiNhatKy(m MucNhatKy) {
	if m.Luc == "" {
		m.Luc = time.Now().Format(time.RFC3339)
	}
	b, err := json.Marshal(m)
	if err != nil {
		return
	}

	nhatKyMu.Lock()
	defer nhatKyMu.Unlock()

	if err := os.MkdirAll(thuMucNhatKy(), 0o700); err != nil {
		return
	}
	ten := filepath.Join(thuMucNhatKy(), time.Now().Format("2006-01")+".jsonl")
	f, err := os.OpenFile(ten, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o600)
	if err != nil {
		return
	}
	defer f.Close()
	f.Write(append(b, '\n'))
}

// DocNhatKy trả về mới nhất trước. thang rỗng = tháng hiện tại.
func DocNhatKy(thang string, gioiHan int) []MucNhatKy {
	if thang == "" {
		thang = time.Now().Format("2006-01")
	}
	b, err := os.ReadFile(filepath.Join(thuMucNhatKy(), thang+".jsonl"))
	if err != nil {
		return nil
	}

	var ds []MucNhatKy
	for _, dong := range strings.Split(string(b), "\n") {
		if strings.TrimSpace(dong) == "" {
			continue
		}
		var m MucNhatKy
		if json.Unmarshal([]byte(dong), &m) == nil {
			ds = append(ds, m)
		}
	}
	sort.SliceStable(ds, func(i, j int) bool { return ds[i].Luc > ds[j].Luc })
	if gioiHan > 0 && len(ds) > gioiHan {
		ds = ds[:gioiHan]
	}
	return ds
}

// locTruongCam dựng chuỗi mô tả form, bỏ mọi trường bí mật.
func locTruongCam(v url.Values) string {
	var phan []string
	khoa := make([]string, 0, len(v))
	for k := range v {
		khoa = append(khoa, k)
	}
	sort.Strings(khoa)
	for _, k := range khoa {
		if truongCam[k] {
			continue
		}
		phan = append(phan, k+"="+catBot(v.Get(k), 60))
	}
	return strings.Join(phan, " ")
}

// boNhatKy ghi mọi thay đổi dữ liệu. Chỉ ghi POST/PUT/PATCH/DELETE — ghi cả
// GET thì mỗi lần Kendy mở trang là một dòng, file ngập trong vài ngày và
// không ai đọc nữa.
func boNhatKy(h http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		theoDoi := r.Method != http.MethodGet && r.Method != http.MethodHead &&
			(strings.HasPrefix(r.URL.Path, "/qt") || strings.HasPrefix(r.URL.Path, "/api"))
		if !theoDoi {
			h.ServeHTTP(w, r)
			return
		}

		// Đọc form TRƯỚC khi handler chạy: handler có thể tiêu thụ hết body.
		r.ParseForm()
		mo := locTruongCam(r.PostForm)

		ai := "khach"
		if nd, ok := NguoiDangNhap(r); ok {
			ai = nd.Ten
		}

		bat := &batMa{ResponseWriter: w, ma: http.StatusOK}
		h.ServeHTTP(bat, r)

		ket := "ok"
		if bat.ma >= 400 {
			ket = "loi-" + strconv.Itoa(bat.ma)
		}
		GhiNhatKy(MucNhatKy{
			Ai: ai, IP: ipCua(r), Viec: "sua-du-lieu",
			Duong: r.URL.Path, DoiTuong: mo, KetQua: ket,
		})
	})
}

// batMa nhớ mã HTTP mà handler đã ghi, để nhật ký phân biệt được việc thành
// công với việc bị từ chối.
type batMa struct {
	http.ResponseWriter
	ma int
}

func (b *batMa) WriteHeader(ma int) {
	b.ma = ma
	b.ResponseWriter.WriteHeader(ma)

package core

// TOTP theo RFC 6238, tự cài bằng thư viện chuẩn.
//
// Vì sao không kéo thư viện OTP: cả app đang zero-dependency, mà phần cần
// dùng chỉ là HMAC-SHA1 rồi cắt 4 byte — dưới 80 dòng. Thư viện đầy đủ kéo
// theo QR encoder, nhiều thuật toán băm, nhiều độ dài mã, không dùng tới.
//
// Không vẽ mã QR: tự viết QR encoder là việc lớn cho 1-3 người dùng. Hiện
// chuỗi base32 để dán tay vào Google Authenticator, kèm URI otpauth:// cho ai
// muốn tự dựng QR.

import (
	"crypto/hmac"
	"crypto/sha1"
	"encoding/base32"
	"encoding/binary"
	"fmt"
	"net/url"
	"strings"
	"time"
)

const buocTOTP = 30 // giây, theo mặc định của Google Authenticator

func maTOTP(biMat []byte, buoc int64) string {
	var b [8]byte
	binary.BigEndian.PutUint64(b[:], uint64(buoc))

	m := hmac.New(sha1.New, biMat)
	m.Write(b[:])
	tong := m.Sum(nil)

	// Cắt động theo RFC 4226 mục 5.3: 4 bit cuối chỉ ra vị trí bắt đầu.
	vt := tong[len(tong)-1] & 0x0f
	so := (uint32(tong[vt])&0x7f)<<24 |
		uint32(tong[vt+1])<<16 |
		uint32(tong[vt+2])<<8 |
		uint32(tong[vt+3])
	return fmt.Sprintf("%06d", so%1000000)
}

// kiemTOTP trả về bước thời gian đã khớp, để chỗ gọi nhớ mà từ chối dùng lại.
// Cửa sổ ±1 bước: đồng hồ điện thoại lệch vài giây là chuyện thường.
func kiemTOTP(biMatB32, ma string, luc time.Time) (int64, bool) {
	ma = strings.TrimSpace(ma)
	if len(ma) != 6 {
		return 0, false
	}
	for _, c := range ma {
		if c < '0' || c > '9' {
			return 0, false
		}
	}

	biMat, err := base32.StdEncoding.WithPadding(base32.NoPadding).
		DecodeString(strings.ToUpper(strings.ReplaceAll(biMatB32, " ", "")))
	if err != nil || len(biMat) == 0 {
		return 0, false
	}

	hienTai := luc.Unix() / buocTOTP
	for d := int64(-1); d <= 1; d++ {
		if hmac.Equal([]byte(maTOTP(biMat, hienTai+d)), []byte(ma)) {
			return hienTai + d, true
		}
	}
	return 0, false
}

func sinhBiMatTOTP() string {
	// 20 byte = 160 bit, đúng độ dài RFC 4226 khuyến nghị cho SHA-1.
	b := make([]byte, 20)
	copy(b, mustHex(maNgauNhien(20)))
	return base32.StdEncoding.WithPadding(base32.NoPadding).EncodeToString(b)
}

func uriOTPAuth(taiKhoan, biMatB32 string) string {
	pht := CFG.ThuongHieu.TenDayDu
	if pht == "" {
		pht = "TramVot"
	}
	return "otpauth://totp/" + url.PathEscape(pht+":"+taiKhoan) +
		"?secret=" + biMatB32 +
		"&issuer=" + url.QueryEscape(pht) +
		"&algorithm=SHA1&digits=6&period=30"
}

func DatTOTP(ten, biMat string, maDuPhong []string) error {
	ten = strings.ToLower(strings.TrimSpace(ten))
	nguoiDungMu.Lock()
	vt := -1
	for i := range nguoiDung {
		if strings.ToLower(nguoiDung[i].Ten) == ten {
			vt = i
			break
		}
	}
	if vt < 0 {
		nguoiDungMu.Unlock()
		return fmt.Errorf("không có tài khoản tên %q", ten)
	}
	var hash []string
	for _, m := range maDuPhong {
		h, err := bcrypt.GenerateFromPassword([]byte(m), 12)
		if err != nil {
			nguoiDungMu.Unlock()
			return err
		}
		hash = append(hash, string(h))
	}
	nguoiDung[vt].TotpBiMat = biMat
	nguoiDung[vt].TotpBat = true
	nguoiDung[vt].MaDuPhong = hash
	nguoiDungMu.Unlock()
	return luuNguoiDung()
}

func TatTOTP(ten string) error {
	ten = strings.ToLower(strings.TrimSpace(ten))
	nguoiDungMu.Lock()
	for i := range nguoiDung {
		if strings.ToLower(nguoiDung[i].Ten) == ten {
			nguoiDung[i].TotpBiMat = ""
			nguoiDung[i].TotpBat = false
			nguoiDung[i].MaDuPhong = nil
		}
	}
	nguoiDungMu.Unlock()
	return luuNguoiDung()
}

// DungMaDuPhong: khớp thì XOÁ mã đó rồi trả true. Xoá là điểm mấu chốt —
// mã dự phòng dùng lại được thì nó chỉ là mật khẩu thứ hai.
func DungMaDuPhong(ten, ma string) bool {
	ten = strings.ToLower(strings.TrimSpace(ten))
	ma = strings.ToUpper(strings.TrimSpace(ma))

	nguoiDungMu.Lock()
	vt, khop := -1, -1
	for i := range nguoiDung {
		if strings.ToLower(nguoiDung[i].Ten) != ten {
			continue
		}
		vt = i
		for j, h := range nguoiDung[i].MaDuPhong {
			if bcrypt.CompareHashAndPassword([]byte(h), []byte(ma)) == nil {
				khop = j
				break
			}
		}
		break
	}
	if vt < 0 || khop < 0 {
		nguoiDungMu.Unlock()
		return false
	}
	ds := nguoiDung[vt].MaDuPhong
	nguoiDung[vt].MaDuPhong = append(ds[:khop], ds[khop+1:]...)
	nguoiDungMu.Unlock()

	// Nhả khoá rồi mới lưu: luuNguoiDung tự RLock, gọi khi đang giữ Lock là
	// treo cả tiến trình — cùng cái bẫy đã gặp ở DoiVaiTro.
	luuNguoiDung()
	return true
}

func sinhMaDuPhong(n int) []string {
	ds := make([]string, n)
	for i := range ds {
		s := strings.ToUpper(maNgauNhien(4)) // 8 ký tự hex
		ds[i] = s[:4] + "-" + s[4:]
	}
	return ds
}

func thanMailDangNhapLa(ten, ip, luc string) (tieuDe, thanHTML string) {
